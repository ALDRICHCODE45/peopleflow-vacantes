 <TabsContent value="summary" className="flex flex-col gap-7"><section className="flex flex-col gap-5"><SectionTitle>Información de la postulación</SectionTitle><dl className="grid grid-cols-[1fr_1fr] items-center gap-x-4 gap-y-5"><dt className="flex items-center gap-2 text-xs text-muted-foreground"><ArrowUpRight className="size-4"/>Etapa actual</dt><dd className="text-sm font-medium">{stages[c.stage].name}</dd><dt className="flex items-center gap-2 text-xs text-muted-foreground"><Clock3 className="size-4"/>En la etapa desde</dt><dd className="text-sm font-medium tabular-nums">{c.since}</dd><dt className="flex items-center gap-2 text-xs text-muted-foreground"><CalendarDays className="size-4"/>Postulación</dt><dd className="text-sm font-medium tabular-nums">17 oct 2026</dd><dt className="flex items-center gap-2 text-xs text-muted-foreground"><Globe2 className="size-4"/>Fuente</dt><dd className="text-sm font-medium">Orgánica</dd></dl></section><Separator/>

 <TabsContent value="answers" className="flex flex-col gap-6"><SectionTitle>Preguntas de filtro</SectionTitle>{[['¿Cuentas con disponibilidad para trabajar en México?','Sí, tengo disponibilidad.'],['¿Cuál es tu expectativa salarial mensual bruta?','$45,000 MXN'],['¿Cuándo podrías incorporarte?','Dos semanas después de recibir una oferta.']].map(([q,a])=><div key={q} className="flex flex-col gap-2 border-b pb-5"><p className="text-[13px] text-muted-foreground">{q}</p><p className="text-sm leading-6">{a}</p></div>)}</TabsContent>

 </>}</TabErrorBoundary></div></Tabs><div className="flex items-center justify-between border-t px-6 py-4 text-xs text-muted-foreground"><span>Software Developer</span><span>Postulación #{String(c.id+1).padStart(4,'0')}</span></div>
 </>}</SheetContent></Sheet>
