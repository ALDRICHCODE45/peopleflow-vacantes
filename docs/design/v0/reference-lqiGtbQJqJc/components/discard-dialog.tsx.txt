'use client'
import {useState} from 'react'
import {UserRoundX, Info} from 'lucide-react'
import {Dialog,DialogContent,DialogHeader,DialogTitle,DialogDescription,DialogFooter} from '@/components/ui/dialog'
import {Select,SelectTrigger,SelectValue,SelectContent,SelectGroup,SelectLabel,SelectItem} from '@/components/ui/select'
import {Field,FieldGroup,FieldLabel} from '@/components/ui/field'
import {Textarea} from '@/components/ui/textarea'
import {Button} from '@/components/ui/button'
import {Candidate,reasonGroups} from '@/lib/pipeline'
export function DiscardDialog({candidate,onClose,onConfirm}:{candidate:Candidate|null;onClose:()=>void;onConfirm:(reason:string,group:string,internal:string,message:string)=>void}){
 const [reason,setReason]=useState<string|null>(null),[internal,setInternal]=useState(''),[message,setMessage]=useState('')
 const [groupIndex,reasonIndex]=(reason??'0:-1').split(':').map(Number)
 const selected=reasonGroups[groupIndex]?.reasons[reasonIndex]
 const valid=!!selected&&(selected!=='Otro'||internal.trim().length>0)
 return <Dialog open={!!candidate} onOpenChange={open=>{if(!open)onClose()}}><DialogContent className="sm:max-w-[520px]"><DialogHeader><div className="mb-3 flex size-10 items-center justify-center rounded-xl bg-status-danger/10 text-status-danger"><UserRoundX className="size-5"/></div><DialogTitle>Descartar a {candidate?.name}</DialogTitle><DialogDescription>Registra el motivo para mantener informado a tu equipo.</DialogDescription></DialogHeader><form onSubmit={e=>{e.preventDefault();if(valid){onConfirm(selected,reasonGroups[groupIndex].name,internal.trim(),message.trim());setReason(null);setInternal('');setMessage('')}}} className="flex flex-col gap-6"><FieldGroup><Field><FieldLabel>Motivo del descarte <span className="text-destructive">*</span></FieldLabel><Select value={reason} onValueChange={v=>setReason(v)}><SelectTrigger aria-label="Motivo del descarte" className="h-10 w-full"><SelectValue placeholder="Selecciona un motivo">{selected}</SelectValue></SelectTrigger><SelectContent>{reasonGroups.map((group,g)=><SelectGroup key={group.name}><SelectLabel>{group.name}</SelectLabel>{group.reasons.map((r,i)=><SelectItem key={i} value={`${g}:${i}`}>{r}</SelectItem>)}</SelectGroup>)}</SelectContent></Select></Field><Field><div className="flex items-center justify-between"><FieldLabel htmlFor="internal">Comentario interno {selected==='Otro'&&<span className="text-destructive">*</span>}</FieldLabel><span className="text-xs tabular-nums text-muted-foreground">{internal.length}/1000</span></div><Textarea id="internal" value={internal} onChange={e=>setInternal(e.target.value)} maxLength={1000} required={selected==='Otro'} placeholder="Agrega contexto para tu equipo…" className="min-h-24 resize-none"/></Field><Field><div className="flex items-center justify-between"><FieldLabel htmlFor="message">Mensaje para el candidato <span className="text-xs font-normal text-muted-foreground">Opcional</span></FieldLabel><span className="text-xs tabular-nums text-muted-foreground">{message.length}/500</span></div><Textarea id="message" value={message} onChange={e=>setMessage(e.target.value)} maxLength={500} placeholder="Escribe un mensaje para el candidato…" className="min-h-24 resize-none"/><p className="flex items-start gap-2 text-xs leading-5 text-muted-foreground"><Info className="mt-0.5 size-3.5 shrink-0"/>El candidato verá este mensaje con el estado «Finalizada».</p></Field></FieldGroup><DialogFooter className="bg-transparent"><Button type="button" variant="outline" onClick={onClose} className="h-10">Cancelar</Button><Button variant="destructive" type="submit" disabled={!valid} className="h-10">Descartar candidato</Button></DialogFooter></form></DialogContent></Dialog>
}
