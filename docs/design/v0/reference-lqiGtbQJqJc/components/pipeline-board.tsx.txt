import { useState } from 'react'
import { ArrowLeft, ArrowRight, BriefcaseBusiness, ChevronRight, ChevronsUpDown, CircleHelp, Columns3, LayoutGrid, LockKeyhole, Moon, Settings2, Sun, UsersRound, Sparkles, PanelLeftClose, CheckCheck, GripVertical, X, Check } from 'lucide-react'
import {Button} from '@/components/ui/button'
import {Badge} from '@/components/ui/badge'
import {Avatar,AvatarFallback} from '@/components/ui/avatar'
import {TooltipProvider} from '@/components/ui/tooltip'
import {Dialog,DialogContent,DialogTitle,DialogDescription,DialogHeader} from '@/components/ui/dialog'
import {CandidateCard,Hint} from './candidate-card'
import {CandidateSheet} from './candidate-sheet'
import {DiscardDialog} from './discard-dialog'
import {allowed,initialCandidates,stages} from '@/lib/pipeline'
import {cn} from '@/lib/utils'
export default function PipelineBoard(){

 const candidate=candidates.find(c=>c.id===selected)??null
 const dragged=candidates.find(c=>c.id===drag)
 const count=candidates.filter(c=>c.stage===4).length
 const [everCovered,setEverCovered]=useState(false)
 const notify=(message:string)=>{setNotice(message);setTimeout(()=>setNotice(''),4000)}
 const move=(id:number,to:number)=>{const c=candidates.find(c=>c.id===id);if(!c||c.moving||to===5||(c.stage!==5&&!allowed(c.stage).includes(to)))return;setCandidates(prev=>prev.map(x=>x.id===id?{...x,moving:true}:x));setTimeout(()=>{setCandidates(prev=>{const next=prev.map(x=>x.id===id?{...x,stage:to,moving:false,since:'19 oct 2026',history:[...x.history,{title:x.stage===5?'Recuperación':`${stages[x.stage].name} → ${stages[to].name}`,date:'19 oct 2026, 10:24',...(to===4?{badge:'Contratación'}:x.stage===4?{badge:'Salida de Placement'}:{})}]}:x);if(next.filter(x=>x.stage===4).length>=2)setEverCovered(true);return next});notify(`${c.name} se movió a ${stages[to].name}`)},650)}
 const toggleTheme=()=>{setDark(!dark);document.documentElement.classList.toggle('dark',!dark)}
 return <TooltipProvider><div className="flex min-h-screen bg-background">
 <aside className="hidden w-[216px] shrink-0 flex-col border-r bg-card md:flex"><div className="flex h-[76px] items-center justify-between px-6"><div className="flex items-center gap-2.5"><span className="flex size-8 items-center justify-center rounded-lg bg-primary text-primary-foreground"><Columns3 className="size-5"/></span><span className="text-[23px] font-semibold tracking-tight">nexo<span className="text-primary">.</span></span></div><Hint text="Espacio de demostración"><span className="text-muted-foreground"><PanelLeftClose className="size-4"/></span></Hint></div><div className="px-4"><div className="flex items-center gap-2.5 rounded-lg border px-3 py-3"><span className="flex size-8 items-center justify-center rounded-md bg-secondary font-semibold">N</span><div className="flex flex-1 flex-col gap-0.5"><span className="text-xs font-medium">Mi empresa</span><span className="text-[11px] text-muted-foreground">Espacio de trabajo</span></div><ChevronsUpDown className="size-3.5 text-muted-foreground"/></div></div><div className="flex flex-col gap-1 px-3 py-7"><p className="px-3 pb-2 text-[10px] font-medium uppercase tracking-widest text-muted-foreground">Reclutamiento</p><div className="flex h-11 items-center gap-3 rounded-lg bg-secondary px-3 font-medium"><BriefcaseBusiness className="size-4"/>Vacantes<span className="ml-auto rounded-md bg-card px-1.5 text-xs text-muted-foreground">1</span></div><div className="flex h-11 items-center gap-3 px-3 text-muted-foreground"><UsersRound className="size-4"/>Candidatos</div></div><div className="mt-auto flex flex-col gap-4 p-4"><div className="rounded-xl bg-background p-3"><div className="mb-2 flex items-center gap-2 text-xs font-medium"><Sparkles className="size-3.5"/>Un mejor siguiente paso</div><p className="text-xs leading-5 text-muted-foreground">Todo tu proceso de selección, en un solo lugar.</p></div><Button variant="ghost" className="h-10 justify-start text-muted-foreground" onClick={()=>setHelp(true)}><CircleHelp/>Ayuda y recursos</Button><div className="flex items-center gap-2.5 border-t pt-4"><Avatar className="size-8"><AvatarFallback className="text-xs">MR</AvatarFallback></Avatar><div className="flex flex-col gap-0.5"><span className="text-xs font-medium">Mariana Rodríguez</span><span className="text-[11px] text-muted-foreground">Reclutador</span></div></div></div></aside>
 <main className="flex min-w-0 flex-1 flex-col"><header className="flex h-[76px] shrink-0 items-center justify-between gap-3 border-b bg-card px-5 lg:px-8"><nav aria-label="Ruta de navegación" className="flex min-w-0 items-center gap-3 text-[13px]"><BriefcaseBusiness className="size-4 shrink-0 text-muted-foreground"/><span className="hidden text-muted-foreground sm:inline">Vacantes</span><ChevronRight className="size-3.5 shrink-0 text-muted-foreground"/><span className="truncate font-medium">Software Developer</span><Badge className="shrink-0 border-0 bg-status-success/10 text-status-success"><span className="size-1.5 rounded-full bg-status-success"/>Activa</Badge></nav><Hint text={dark?'Cambiar a tema claro':'Cambiar a tema oscuro'}><Button variant="ghost" size="icon" className="size-10 shrink-0" onClick={toggleTheme} aria-label={dark?'Cambiar a tema claro':'Cambiar a tema oscuro'}>{dark?<Sun/>:<Moon/>}</Button></Hint></header>
 <div className="flex flex-wrap items-center justify-between gap-4 px-5 py-7 lg:px-8"><div className="flex flex-col gap-1.5"><h1 className="text-xl font-semibold tracking-tight">Pipeline</h1><p className="text-xs text-muted-foreground">Publicada el 17 oct 2026</p></div><div className="flex items-center gap-5"><div className="flex min-w-40 flex-col gap-2"><div className="flex items-center gap-2 text-xs"><CheckCheck className="size-4 text-status-success"/>{count>=2?<span className="font-medium text-status-success">Cobertura completa</span>:<span><strong className="font-semibold tabular-nums">{count} de 2</strong> plazas cubiertas</span>}</div><div role="progressbar" aria-label="Plazas cubiertas" aria-valuenow={count} aria-valuemin={0} aria-valuemax={2} className="h-1 w-full overflow-hidden rounded-full bg-border"><div className="h-full rounded-full bg-status-success transition-all" style={{width:`${Math.min(count/2,1)*100}%`}}/></div>{(everCovered||count>=2)&&<span className="text-[11px] text-muted-foreground">{count>=2?'19 oct 2026':'Se completó el 19 oct 2026'}</span>}</div><span className="h-8 w-px bg-border"/><Hint text="Configuración de la vacante"><Button variant="outline" size="icon" className="size-10 bg-card" onClick={()=>setSettings(true)} aria-label="Configuración de la vacante"><Settings2/></Button></Hint></div></div>
 <div className="flex items-center justify-between gap-4 px-5 pb-4 lg:px-8"><div className="flex items-center gap-2 text-xs text-muted-foreground"><UsersRound className="size-4"/><span><strong className="font-medium text-foreground">9</strong> candidatos en el pipeline</span></div><span className="hidden items-center gap-1.5 text-[11px] text-muted-foreground sm:flex"><GripVertical className="size-3.5"/>Arrastra una tarjeta para moverla</span></div>
 <div className="board-scroll mx-5 snap-x snap-mandatory overflow-x-auto pb-6 lg:mx-8"><div className="flex w-max items-start gap-3 pb-3">{stages.map((stage,index)=>{const list=candidates.filter(c=>c.stage===index);const accepts=dragged&&allowed(dragged.stage).includes(index);return <section key={stage.name} aria-label={stage.name} onDragOver={e=>{if(accepts)e.preventDefault()}} onDrop={e=>{e.preventDefault();if(accepts&&drag!==null)move(drag,index);setDrag(null)}} className={cn('stage-column w-[284px] shrink-0 snap-start rounded-xl p-2 transition-opacity',stage.color,dragged&&!accepts&&dragged.stage!==index&&'opacity-35',accepts&&'outline-1 outline-dashed outline-primary')}><div className="stage-heading mb-3 flex h-10 items-center gap-2 rounded-lg px-3"><span className="stage-dot size-2 rounded-full"/><h2 className="text-[13px] font-semibold">{stage.name}</h2><Badge className="ml-1 border-0 bg-card/60 px-1.5 text-[11px] tabular-nums text-inherit">{list.length}</Badge>{index===5&&<Hint text="Se llega descartando, no moviendo"><span tabIndex={0} className="ml-auto flex size-8 items-center justify-center"><LockKeyhole className="size-3.5"/></span></Hint>}</div><div className="flex flex-col gap-3">{list.map(c=><CandidateCard key={c.id} candidate={c} onOpen={()=>setSelected(c.id)} onMove={n=>move(c.id,n)} onDiscard={()=>setDiscard(c.id)} onDrag={setDrag}/>)}{list.length===0&&<div className="flex h-24 items-center justify-center rounded-lg border border-dashed border-border text-[11px] text-muted-foreground">Sin candidatos en esta etapa</div>}</div></section>})}</div></div>
 <div className="flex flex-wrap items-center gap-2 px-5 py-2 text-[11px] text-muted-foreground lg:px-8"><LockKeyhole className="size-3.5"/><span>Tu proceso, a tu ritmo. Los cambios son visibles solo para tu equipo.</span></div><footer className="mt-auto flex items-center justify-between gap-3 px-5 py-5 text-[11px] text-muted-foreground lg:px-8"><span>Espacio de demostración · Los cambios no se guardan</span><Button variant="ghost" className="h-10 text-xs" onClick={()=>setSelected(0)}>Ver ficha de ejemplo <ArrowRight/></Button></footer>
 </main></div><CandidateSheet candidate={candidate} onClose={()=>setSelected(null)} onMove={n=>selected!==null&&move(selected,n)} onDiscard={()=>setDiscard(selected)}/><DiscardDialog key={discard??'closed'} candidate={candidates.find(c=>c.id===discard)??null} onClose={()=>setDiscard(null)} onConfirm={(reason,group,internal,message)=>{setCandidates(prev=>prev.map(c=>c.id===discard?{...c,previous:c.stage,stage:5,since:'19 oct 2026',reason,group,internal,message,history:[...c.history,{title:'Descarte',date:'19 oct 2026, 10:30',...(c.stage===4?{badge:'Salida de Placement'}:{})}]}:c));setDiscard(null);notify('Candidato descartado. Puedes recuperarlo desde su ficha.')}}/>
 <Dialog open={settings} onOpenChange={setSettings}><DialogContent><DialogHeader><DialogTitle>Configuración de la vacante</DialogTitle><DialogDescription>Software Developer · Vacante activa</DialogDescription></DialogHeader><dl className="grid grid-cols-2 gap-4 py-4"><dt className="text-muted-foreground">Plazas disponibles</dt><dd>2 plazas</dd><dt className="text-muted-foreground">Publicación</dt><dd>17 oct 2026</dd><dt className="text-muted-foreground">Etapas</dt><dd>5 etapas + descartados</dd></dl><p className="text-xs text-muted-foreground">Configuración de solo lectura en esta demostración.</p></DialogContent></Dialog>
 <Dialog open={help} onOpenChange={setHelp}><DialogContent><DialogHeader><DialogTitle>Tu pipeline, paso a paso</DialogTitle><DialogDescription>Explora el proceso con datos de ejemplo.</DialogDescription></DialogHeader><div className="flex flex-col gap-4 leading-6"><p>Abre una tarjeta para consultar la ficha, las respuestas y el historial.</p><p>Arrastra un candidato a la siguiente etapa o a cualquiera anterior. También puedes usar el menú de acciones.</p><p>Para descartar, selecciona un motivo. Podrás recuperar al candidato más adelante.</p></div></DialogContent></Dialog>
 {notice&&<div role="status" className="fixed bottom-6 left-1/2 flex max-w-[90vw] -translate-x-1/2 items-center gap-3 rounded-xl border bg-popover px-4 py-3 text-sm shadow-lg"><Check className="size-4 shrink-0 text-status-success"/>{notice}<Button variant="ghost" size="icon" aria-label="Ocultar notificación" onClick={()=>setNotice('')}><X/></Button></div>}
 </TooltipProvider>
}
