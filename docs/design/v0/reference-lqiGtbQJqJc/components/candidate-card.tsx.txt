'use client'
import { ArrowRight, CalendarDays, Clock3, FileText, FileType2, GripVertical, MoreHorizontal, RotateCcw, UserRound, UserRoundX, LoaderCircle } from 'lucide-react'
import { Button } from '@/components/ui/button'
import { Avatar, AvatarFallback } from '@/components/ui/avatar'
import { Badge } from '@/components/ui/badge'
import { Separator } from '@/components/ui/separator'
import { Tooltip, TooltipContent, TooltipTrigger } from '@/components/ui/tooltip'
import { DropdownMenu, DropdownMenuContent, DropdownMenuGroup, DropdownMenuItem, DropdownMenuLabel, DropdownMenuSeparator, DropdownMenuTrigger } from '@/components/ui/dropdown-menu'
import { allowed, Candidate, initials, stages } from '@/lib/pipeline'
import { cn } from '@/lib/utils'
export function Hint({text,children}:{text:string;children:React.ReactElement}) {return <Tooltip><TooltipTrigger render={children}/><TooltipContent>{text}</TooltipContent></Tooltip>}
export function CandidateMenu({candidate,onMove,onOpen,onDiscard,full=false}:{candidate:Candidate;onMove:(n:number)=>void;onOpen:()=>void;onDiscard:()=>void;full?:boolean}){
 return <DropdownMenu><DropdownMenuTrigger render={<Button variant="outline" size={full?'default':'icon'} className={cn('h-10',!full&&'border-transparent bg-transparent shadow-none')} aria-label={full?'Mover a…':`Acciones de ${candidate.name}`} onClick={e=>e.stopPropagation()}/>} >{full?<><ArrowRight/>Mover a…</>:<MoreHorizontal/>}</DropdownMenuTrigger><DropdownMenuContent align="end" className="w-56" onClick={e=>e.stopPropagation()}><DropdownMenuGroup>{candidate.stage===5?<DropdownMenuItem onClick={()=>onMove(candidate.previous??0)}><RotateCcw/>Recuperar</DropdownMenuItem>:<><DropdownMenuLabel>Mover a…</DropdownMenuLabel>{allowed(candidate.stage).map(i=><DropdownMenuItem key={i} onClick={()=>onMove(i)}><span className={cn('size-2 rounded-full stage-dot',stages[i].color)}/>{stages[i].name}<ArrowRight className="ml-auto"/></DropdownMenuItem>)}</>}</DropdownMenuGroup>{!full&&<><DropdownMenuSeparator/><DropdownMenuGroup><DropdownMenuItem onClick={onOpen}><UserRound/>Ver ficha</DropdownMenuItem>{candidate.stage!==5&&<DropdownMenuItem variant="destructive" onClick={onDiscard}><UserRoundX/>Descartar</DropdownMenuItem>}</DropdownMenuGroup></>}</DropdownMenuContent></DropdownMenu>
}
export function CandidateCard({candidate:c,onOpen,onMove,onDiscard,onDrag}:{candidate:Candidate;onOpen:()=>void;onMove:(n:number)=>void;onDiscard:()=>void;onDrag:(id:number|null)=>void}){
 const FileIcon=c.file.endsWith('pdf')?FileText:FileType2
 return <article role="button" tabIndex={c.moving?-1:0} aria-label={`Ver ficha de ${c.name}`} aria-disabled={c.moving} draggable={!c.moving&&c.stage!==5} onDragStart={e=>{e.dataTransfer.setData('text/plain',String(c.id));onDrag(c.id)}} onDragEnd={()=>onDrag(null)} onClick={()=>!c.moving&&onOpen()} onKeyDown={e=>{if(e.target===e.currentTarget&&(e.key==='Enter'||e.key===' ')){e.preventDefault();!c.moving&&onOpen()}}} className={cn('candidate-card group relative flex flex-col gap-3 rounded-xl bg-card p-3 transition-shadow hover:shadow-md',c.moving&&'pointer-events-none opacity-55')}>
 <div className="flex items-center gap-2"><Avatar className="size-8 shrink-0"><AvatarFallback className="bg-secondary text-xs font-medium text-muted-foreground">{initials(c.name)}</AvatarFallback></Avatar><Hint text={c.name}><span className="line-clamp-2 min-w-0 flex-1 text-sm font-medium leading-5">{c.name}</span></Hint>{c.moving?<LoaderCircle className="m-2 size-4 animate-spin text-muted-foreground"/>:<CandidateMenu candidate={c} onMove={onMove} onOpen={onOpen} onDiscard={onDiscard}/>}</div>
 <div className="flex flex-col gap-1.5 text-xs text-muted-foreground"><span className="flex items-center gap-2"><CalendarDays className="size-3.5"/>Postuló 17 oct</span><span className="flex items-center gap-2"><Clock3 className="size-3.5"/>En la etapa desde {c.since.replace(' 2026','')}</span></div>
 <Separator/>{c.stage===5?<div className="flex flex-col items-start gap-2"><Badge className="border-0 bg-status-danger/10 text-status-danger text-[11px]">{c.reason}</Badge><span className="text-xs text-muted-foreground">{c.group}</span></div>:<div className="flex items-center gap-2"><span className="flex min-w-0 items-center gap-1.5 rounded-md bg-muted/60 px-2 py-1 text-xs text-muted-foreground"><FileIcon className="size-3.5 shrink-0"/><Hint text={c.file}><span className="truncate">{c.file}</span></Hint></span><GripVertical className="ml-auto size-4 shrink-0 text-muted-foreground opacity-0 group-hover:opacity-100 group-focus-within:opacity-100"/></div>}
 </article>
}
