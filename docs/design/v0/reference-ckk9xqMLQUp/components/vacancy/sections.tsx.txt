export type SectionProps={v:Vacancy;patch:(p:Partial<Vacancy>)=>void;errors:Record<string,string>};
export function BasicSection({v,patch,errors}:SectionProps){
 const [note,setNote]=useState(''),[newName,setNewName]=useState(''),[create,setCreate]=useState(false),[page,setPage]=useState(5);
 const [catalog,setCatalog]=useState<Record<string,string[]>>({'Área interna':['Producto','Diseño','Ingeniería','Marketing','Operaciones','Finanzas','Recursos humanos'],Cliente:['Acme México','Estudio Norte','Grupo Meridiano','Nubia','Horizonte Digital','Luma']});
 const f=(key:keyof Vacancy,label:string,options?:string[],required=false,search=false)=><FormField id={key} label={label} value={String(v[key])} onChange={x=>patch({[key]:x})} error={errors[key]} options={options} required={required} search={search}/>;
 const resolve=()=>{if(!v.postal)return;const p=postalPlaces[v.postal];if(p){patch({state:p.state,city:p.cities.length===1?p.cities[0]:''});setNote(p.cities.length>1?'Este código postal corresponde a varios municipios':'Completamos la ubicación. Puedes cambiarla.')}else setNote('No encontramos ese código postal.')};
 return <div className="flex flex-col gap-5"><Card><CardHeader><CardTitle>El puesto que estás buscando</CardTitle><CardDescription>Empieza por lo esencial para conectar con las personas correctas.</CardDescription><CardAction><BriefcaseBusiness className="size-5 text-muted-foreground"/></CardAction></CardHeader><CardContent><FieldGroup><FormField id="title" label="Título de la vacante" value={v.title} onChange={title=>patch({title})} maxLength={200} placeholder="Ej. Diseñador/a de producto" required error={errors.title} hint="Usa un título claro y fácil de encontrar."/><FieldGroup className="form-grid">{f('area','Área profesional',areas,true,true)}{f('level','Nivel',['Prácticas','Junior','Intermedio','Senior','Líder'])}{f('schedule','Jornada',['Tiempo completo','Medio tiempo'],true)}{f('contract','Tipo de contrato',['Por proyecto','Indefinido','Servicios profesionales','Temporal','Becas'],true)}</FieldGroup></FieldGroup></CardContent></Card>
 <Card><CardHeader><CardTitle>¿Dónde trabajará?</CardTitle><CardDescription>Define la modalidad y la ubicación de la vacante.</CardDescription><CardAction><MapPin className="size-5 text-muted-foreground"/></CardAction></CardHeader><CardContent><FieldGroup>{f('modality','Modalidad',['Presencial','Remoto','Híbrido'],true)}
 {v.modality==='Remoto'&&<><FormField id="scope" label="¿Desde dónde pueden postular?" value={v.scope} onChange={scope=>patch({scope})} options={['Desde cualquier lugar','Solo desde estos países']} required error={errors.scope}/>{v.scope==='Solo desde estos países'&&<Field><FieldLabel htmlFor="countries">Países <span className="text-primary">*</span></FieldLabel><Combobox multiple items={countries} value={v.countries} onValueChange={c=>{if(c.length<=50)patch({countries:c})}}><ComboboxChips><ComboboxValue>{v.countries.map(c=><ComboboxChip key={c}>{c}</ComboboxChip>)}</ComboboxValue><ComboboxChipsInput id="countries" placeholder="Agregar país…"/></ComboboxChips><ComboboxContent><ComboboxEmpty>Sin resultados.</ComboboxEmpty><ComboboxList>{(c:string)=><ComboboxItem key={c} value={c}>{c}</ComboboxItem>}</ComboboxList></ComboboxContent></Combobox>{errors.countries&&<FieldError>{errors.countries}</FieldError>}</Field>}<div className="flex items-center justify-between"><span className="text-sm font-medium">Ubicación (opcional)</span><Button variant="ghost" onClick={()=>patch({country:'',state:'',city:'',postal:''})}>Quitar ubicación</Button></div></>}
 <Separator/><FieldGroup className="form-grid"><FormField id="country" label="País" value={v.country} onChange={country=>{patch({country,state:'',city:'',postal:''});setNote('')}} options={countries} search required={v.modality!=='Remoto'} error={errors.country}/>{v.country==='México'&&<FormField id="postal" label="Código postal" value={v.postal} onChange={postal=>{patch({postal});setNote('')}} onBlur={resolve} placeholder="Ej. 06700" maxLength={5} hint="Completamos tu ubicación por ti." error={errors.postal}/>}</FieldGroup>
 {v.country==='México'&&<FieldGroup className="form-grid"><FormField id="state" label="Estado" value={v.state} onChange={state=>{patch({state,city:'',postal:''});setNote('')}} options={Object.keys(locations)} search required={v.modality!=='Remoto'} error={errors.state}/><FormField id="city" label="Municipio" value={v.city} onChange={city=>{const match=postalPlaces[v.postal];if(v.postal&&match&&!match.cities.includes(city)){patch({city,postal:''});setNote('Quitamos el código postal porque no corresponde al municipio elegido.')}else patch({city})}} options={locations[v.state]||[]} search disabled={!v.state} required={v.modality!=='Remoto'} error={errors.city}/></FieldGroup>}
 {note&&<p role="status" className="text-sm text-muted-foreground">{note}</p>}{v.postal&&postalPlaces[v.postal]?.cities.length>1&&<div className="flex flex-wrap gap-2">{postalPlaces[v.postal].cities.map(city=><Button key={city} variant="outline" onClick={()=>patch({city})}>{city}</Button>)}</div>}{v.country&&v.country!=='México'&&<p className="text-sm text-muted-foreground">Por ahora solo detallamos ubicaciones en México.</p>}
 </FieldGroup></CardContent></Card>
 <Card><CardHeader><CardTitle>Solicitante</CardTitle><CardDescription>¿Quién necesita cubrir esta posición?</CardDescription><CardAction><Badge variant="secondary"><LockKeyhole/> Solo interno</Badge></CardAction></CardHeader><CardContent><FieldGroup><FieldGroup className="form-grid"><FormField id="requesterType" label="Tipo de solicitante" value={v.requesterType} onChange={requesterType=>{patch({requesterType,requester:''});setPage(5);toast('Quitamos la selección porque cambiaste el tipo de solicitante.')}} options={['Área interna','Cliente']}/><FormField id="requester" label="Área o cliente" value={v.requester} onChange={requester=>patch({requester})} options={catalog[v.requesterType].slice(0,page)} search required error={errors.requester}/></FieldGroup><div className="flex flex-wrap gap-2">{page<catalog[v.requesterType].length&&<Button variant="ghost" onClick={()=>setPage(page+5)}>Cargar más</Button>}{v.requester&&<Button variant="ghost" onClick={()=>patch({requester:''})}>Quitar selección</Button>}<Button variant="ghost" onClick={()=>setCreate(true)}><Plus/>Crear área o cliente</Button></div></FieldGroup></CardContent></Card>
 <Dialog open={create} onOpenChange={setCreate}><DialogContent><DialogHeader><DialogTitle>Crear área o cliente</DialogTitle><DialogDescription>Se creará de inmediato en este catálogo de demostración, independientemente del borrador.</DialogDescription></DialogHeader><FormField id="new-requester" label="Nombre" value={newName} onChange={setNewName} maxLength={160}/>{newName&&<p className="text-sm">Crear “{newName}”</p>}<DialogFooter><DialogClose render={<Button variant="outline"/>}>Cancelar</DialogClose><Button disabled={!newName.trim()} onClick={()=>{const existing=catalog[v.requesterType].find(n=>n.toLowerCase()===newName.trim().toLowerCase());const name=existing||newName.trim();if(!existing)setCatalog({...catalog,[v.requesterType]:[name,...catalog[v.requesterType]]});patch({requester:name});setCreate(false);setNewName('')}}>{catalog[v.requesterType].some(n=>n.toLowerCase()===newName.trim().toLowerCase())?'Seleccionar existente':'Crear y seleccionar'}</Button></DialogFooter></DialogContent></Dialog>
 </div>
}
export function ProfileSection({v,patch,errors}:SectionProps){return <div className="flex flex-col gap-5"><Card><CardHeader><CardTitle>Cuéntales sobre la oportunidad</CardTitle><CardDescription>Una buena descripción ayuda a imaginar cómo será trabajar aquí.</CardDescription></CardHeader><CardContent><FieldGroup><Field data-invalid={!!errors.description}><FieldLabel htmlFor="description">Descripción <span className="text-primary">*</span></FieldLabel><RichEditor value={v.description} onChange={description=>patch({description})} error={errors.description}/>{errors.description&&<FieldError>{errors.description}</FieldError>}</Field><FormField id="required" label="Requisitos obligatorios" value={v.required} onChange={required=>patch({required})} multiline placeholder="Lo indispensable para desempeñar el puesto."/><FormField id="desired" label="Requisitos deseables" value={v.desired} onChange={desired=>patch({desired})} multiline placeholder="Lo que suma, sin ser obligatorio."/></FieldGroup></CardContent></Card><Card><CardHeader><CardTitle>Habilidades e idiomas</CardTitle><CardDescription>Define los conocimientos que aportarán al equipo.</CardDescription></CardHeader><CardContent><FieldGroup><EntryList label="Habilidades" items={v.skills} onChange={skills=>patch({skills})} placeholder="Ej. Diseño de interfaces"/><Separator/><FieldSet><FieldLegend>Idiomas</FieldLegend>{v.languages.map((l,i)=><FieldGroup key={l.id} className="rounded-lg border p-4"><div className="flex items-center justify-between"><span className="text-sm">Idioma {i+1}</span><div className="flex"><Reorder index={i} total={v.languages.length} onMove={d=>{const a=[...v.languages];[a[i],a[i+d]]=[a[i+d],a[i]];patch({languages:a})}}/><Button variant="ghost" size="icon" aria-label={`Eliminar idioma ${i+1}`} onClick={()=>patch({languages:v.languages.filter((_,j)=>j!==i)})}><Trash2/></Button></div></div><FieldGroup className="form-grid"><FormField id={`language-${i}`} label="Idioma" value={l.name} options={['Español','Inglés','Otro']} onChange={name=>patch({languages:v.languages.map(x=>x.id===l.id?{...x,name}:x)})} error={errors[`language-${i}`]}/><FormField id={`language-level-${i}`} label="Nivel" value={l.level} options={['Básico','Intermedio','Avanzado','Nativo']} onChange={level=>patch({languages:v.languages.map(x=>x.id===l.id?{...x,level}:x)})}/></FieldGroup>{l.name==='Otro'&&<FormField id={`language-other-${i}`} label="Nombre del idioma" value={l.other} onChange={other=>patch({languages:v.languages.map(x=>x.id===l.id?{...x,other}:x)})}/>}</FieldGroup>)}<Button variant="outline" className="self-start" disabled={v.languages.length>=100} onClick={()=>patch({languages:[...v.languages,{id:crypto.randomUUID(),name:'',other:'',level:''}]})}><Plus/>Agregar idioma</Button></FieldSet></FieldGroup></CardContent></Card></div>}

export function QuestionsSection({v,patch,errors}:SectionProps){
 const [simulate,setSimulate]=useState(false),[step,setStep]=useState(0),[answers,setAnswers]=useState<Record<number,string>>({});
 const add=()=>patch({questions:[...v.questions,{id:crypto.randomUUID(),text:'',type:'Abierta (respuesta libre)',options:['',''],correct:-1}]});
 const current=v.questions[step];
 return <><Card><CardHeader><CardTitle>Conoce mejor a quienes se postulan</CardTitle><CardDescription>Preguntas de filtro opcionales · {v.questions.length} de 100</CardDescription><CardAction><Badge variant="secondary">Opcional</Badge></CardAction></CardHeader><CardContent><FieldGroup>{!v.questions.length?<Empty><EmptyHeader><EmptyMedia variant="icon"><MessageSquare/></EmptyMedia><EmptyTitle>La conversación empieza aquí</EmptyTitle><EmptyDescription>Agrega preguntas para conocer la experiencia y las expectativas de cada persona. También puedes continuar sin ellas.</EmptyDescription></EmptyHeader><EmptyContent><Button variant="outline" onClick={add}><Plus/>Agregar pregunta</Button></EmptyContent></Empty>:v.questions.map((q,i)=>{const update=(p:Partial<typeof q>)=>patch({questions:v.questions.map(x=>x.id===q.id?{...x,...p}:x)});return <FieldGroup key={q.id} className="rounded-xl border p-4"><div className="flex items-center justify-between"><Badge variant="secondary">Pregunta {i+1}</Badge><div className="flex"><Reorder index={i} total={v.questions.length} onMove={d=>{const a=[...v.questions];[a[i],a[i+d]]=[a[i+d],a[i]];patch({questions:a})}}/><Button variant="ghost" size="icon" aria-label={`Eliminar pregunta ${i+1}`} onClick={()=>patch({questions:v.questions.filter(x=>x.id!==q.id)})}><Trash2/></Button></div></div><FormField id={`question-${i}`} label="Enunciado" value={q.text} onChange={text=>update({text})} maxLength={2000} error={errors[`question-${i}`]} placeholder="Ej. ¿Qué proyecto te enorgullece más?"/><FormField id={`question-type-${i}`} label="Tipo" value={q.type} onChange={type=>update({type,correct:-1})} options={['Abierta (respuesta libre)','Opción única (informativa)','Opción única (con respuesta correcta)']}/>{q.type!=='Abierta (respuesta libre)'&&<>{q.options.map((o,j)=><div className="flex items-end gap-2" key={j}><FormField id={`option-${i}-${j}`} label={`Opción ${j+1}`} value={o} onChange={text=>update({options:q.options.map((x,k)=>j===k?text:x)})}/><Button variant="ghost" size="icon" disabled={q.options.length<=2} aria-label={`Eliminar opción ${j+1}`} onClick={()=>update({options:q.options.filter((_,k)=>k!==j),correct:-1})}><Trash2/></Button></div>)}<Button variant="outline" disabled={q.options.length>=8} onClick={()=>update({options:[...q.options,'']})}><Plus/>Agregar opción</Button>{q.type==='Opción única (con respuesta correcta)'&&<FormField id={`correct-${i}`} label="Respuesta correcta" value={q.correct>=0?`Opción ${q.correct+1}`:''} onChange={s=>update({correct:Number(s.split(' ')[1])-1})} options={q.options.map((_,j)=>`Opción ${j+1}`)} hint="Esta respuesta nunca se muestra a quienes se postulan."/>}</>}</FieldGroup>})}{v.questions.length>0&&<div className="flex flex-wrap gap-3"><Button variant="outline" disabled={v.questions.length>=100} onClick={add}><Plus/>Agregar pregunta</Button><Button variant="outline" onClick={()=>{setStep(0);setAnswers({});setSimulate(true)}}><Play/>Probar cuestionario</Button></div>}</FieldGroup></CardContent></Card><Dialog open={simulate} onOpenChange={setSimulate}><DialogContent><DialogHeader><DialogTitle>Probar cuestionario</DialogTitle><DialogDescription>Simulación · No se guardan respuestas ni se calcula una puntuación.</DialogDescription></DialogHeader>{current&&<FieldGroup><p className="text-sm text-muted-foreground">Pregunta {step+1} de {v.questions.length}</p><FormField id="simulation-answer" label={current.text||'Pregunta sin enunciado'} value={answers[step]||''} onChange={a=>setAnswers({...answers,[step]:a})} multiline={current.type==='Abierta (respuesta libre)'} options={current.type==='Abierta (respuesta libre)'?undefined:current.options.filter(Boolean)}/></FieldGroup>}<DialogFooter><Button variant="outline" disabled={step===0} onClick={()=>setStep(step-1)}>Anterior</Button><Button variant="ghost" onClick={()=>step<v.questions.length-1?setStep(step+1):setSimulate(false)}>Omitir</Button><Button onClick={()=>step<v.questions.length-1?setStep(step+1):setSimulate(false)}>{step===v.questions.length-1?'Finalizar simulación':'Siguiente'}</Button></DialogFooter></DialogContent></Dialog></>
}
