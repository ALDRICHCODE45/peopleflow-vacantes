export function missingFields(v:Vacancy):Missing[] {
 const m:Missing[]=[]; const add=(key:keyof Vacancy,label:string,section=0,group='Información de la vacante')=>m.push({key,label,section,group});
 for(const [k,l] of [['title','Título'],['modality','Modalidad'],['schedule','Jornada'],['contract','Tipo de contrato'],['area','Área profesional'],['requester','Solicitante']] as const) if(!v[k].trim())add(k,l);
 if(!plainText(v.description))add('description','Descripción',1);
 if(v.modality==='Remoto'){if(!v.scope || (v.scope==='Solo desde estos países'&&!v.countries.length))add('scope','Indica desde dónde pueden postular.');}
 else if(v.modality){if(!v.country)add('country','Agrega la ubicación de la vacante.');else if(v.country==='México'&&(!v.state||!v.city))add(!v.state?'state':'city','Completa estado y municipio.');}
 for(const [k,l] of [['requestDate','Fecha de solicitud'],['positions','Plazas a cubrir']] as const)if(!v[k])add(k,l,4,'Configuración de solicitud');
 for(const [k,l] of [['cutoff','Columna de corte'],['target','Objetivo de entrega'],['deadline','Fecha compromiso']] as const)if(!v[k])add(k,l,4,'Configuración de entrega');
 if(v.deadline&&v.deadline<new Date().toISOString().slice(0,10))add('deadline','La fecha compromiso ya pasó.',4,'Configuración de entrega');return m;
}
export function validate(v:Vacancy):Record<string,string>{
 const e:Record<string,string>={};
 if(v.title.length>200)e.title='Usa hasta 200 caracteres.';
 if(v.postal&&!/^\d{5}$/.test(v.postal))e.postal='Escribe exactamente 5 dígitos.';
 else if(v.postal&&!postalPlaces[v.postal])e.postal='No encontramos ese código postal.';
 if(v.state&&!locations[v.state])e.state='Selecciona un estado del catálogo.';
 if(v.city&&!locations[v.state]?.includes(v.city))e.city='Selecciona un municipio de este estado.';
 if(v.countries.length>50)e.countries='Selecciona hasta 50 países.';
 for(const k of ['minimum','maximum'] as const) if(v[k]&&(!/^\d+(\.\d{1,2})?$/.test(v[k])||Number(v[k])>999999999999.99))e[k]='Ingresa un monto válido con hasta 2 decimales.';
 if(v.compensation==='Rango'&&v.maximum&&v.minimum&&Number(v.maximum)<Number(v.minimum))e.maximum='El máximo debe ser igual o mayor que el mínimo.';
 for(const k of ['positions','target'] as const)if(v[k]&&(!/^\d+$/.test(v[k])||Number(v[k])<1))e[k]='Escribe un número entero mayor que cero.';
 v.languages.forEach((l,i)=>{if(!l.name||(l.name==='Otro'&&!l.other.trim()))e[`language-${i}`]='Selecciona o escribe el idioma.';});
 v.questions.forEach((q,i)=>{if(!q.text.trim()||q.text.length>2000)e[`question-${i}`]='Escribe un enunciado de hasta 2,000 caracteres.';else if(q.type!=='Abierta (respuesta libre)'&&(q.options.length<2||q.options.length>8||q.options.some(o=>!o.trim())||new Set(q.options.map(o=>o.trim().toLowerCase())).size!==q.options.length))e[`question-${i}`]='Agrega entre 2 y 8 opciones distintas, sin dejar opciones vacías.';else if(q.type==='Opción única (con respuesta correcta)'&&(q.correct<0||q.correct>=q.options.length))e[`question-${i}`]='Selecciona una respuesta correcta.';});
 return e;
}
